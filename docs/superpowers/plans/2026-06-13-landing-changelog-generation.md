# Landing Changelog Generation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bake real GitHub commits into the static landing changelog at release time via a manual GitHub Actions workflow, and add client-side "load more" pagination — keeping the hand-rolled, zero-dependency site with no runtime GitHub calls.

**Architecture:** A Node generator (`scripts/landing/gen-changelog.mjs`) fetches commits for a release tag from the GitHub API, parses them as conventional commits, and rewrites a marker-delimited `RELEASES` region in `landing/changelog.html` (prepending one entry, idempotent). A manual `workflow_dispatch` workflow runs it with a required `summary` input, commits the result back to `main` with the default `GITHUB_TOKEN` (which never retriggers a workflow), then triggers the existing Render static-site deploy. The changelog render loop gains a "load more" that reveals already-baked entries — pure client-side, no network.

**Tech Stack:** Node 22 (global `fetch`, ESM), Vitest + jsdom (already configured), GitHub Actions, Render API.

```js
// Build-time changelog generator for the Orrery landing page.
// Pure helpers below do all data shaping and are unit-tested; main() (Task 4)
// adds GitHub API IO, CLI parsing and the file write.

export const START_LINE =
  '/* RELEASES:START — generated by scripts/landing/gen-changelog.mjs; do not hand-edit */';
export const END_LINE = '/* RELEASES:END */';

const MONTHS = ['January', 'February', 'March', 'April', 'May', 'June',
  'July', 'August', 'September', 'October', 'November', 'December'];

const CONVENTIONAL_RE = /^(\w+)(?:\(([^)]*)\))?!?:\s*(.+)$/;

export function parseConventionalCommit(subject) {
  const m = CONVENTIONAL_RE.exec(String(subject).trim());
  if (!m) return { type: 'chore', scope: '', msg: String(subject).trim() };
  return { type: m[1].toLowerCase(), scope: (m[2] || '').trim(), msg: m[3].trim() };
}

export function isNoiseCommit(subject) {
  const s = String(subject).trim();
  return s.startsWith('Merge ') || /^release(\(|:)/i.test(s);
}

export function formatDate(iso) {
  const d = new Date(iso);
  return `${MONTHS[d.getUTCMonth()]} ${d.getUTCDate()}, ${d.getUTCFullYear()}`;
}

export function shortHash(sha) {
  return String(sha).slice(0, 7);
}

export function buildReleaseEntry({ tag, channel, date, ref, summary, commits }) {
  return {
    tag,
    channel,
    date,
    ref: shortHash(ref),
    summary,
    commits: commits.map((c) => ({
      type: c.type,
      hash: shortHash(c.hash),
      scope: c.scope || '',
      msg: c.msg,
    })),
  };
}

export function readReleases(html) {
  const start = html.indexOf(START_LINE);
  const end = html.indexOf(END_LINE);
  if (start === -1 || end === -1 || end < start) {
    throw new Error('RELEASES markers not found in changelog.html');
  }
  const region = html.slice(start + START_LINE.length, end);
  const open = region.indexOf('[');
  const close = region.lastIndexOf(']');
  if (open === -1 || close === -1) {
    throw new Error('RELEASES array literal not found between markers');
  }
  return JSON.parse(region.slice(open, close + 1));
}

export function injectRelease(html, entry) {
  const existing = readReleases(html).filter((r) => r.tag !== entry.tag);
  const next = [entry, ...existing];
  const start = html.indexOf(START_LINE);
  const end = html.indexOf(END_LINE);
  const before = html.slice(0, start + START_LINE.length);
  const after = html.slice(end);
  const literal = `\n  const RELEASES = ${JSON.stringify(next, null, 2)};\n  `;
  return before + literal + after;
}
```

```js
const OWNER = 'kouji-dev';
const REPO = 'orrery';
const API = 'https://api.github.com';

function parseArgs(argv) {
  const out = { channel: 'dev', dryRun: false };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === '--summary') out.summary = argv[++i];
    else if (a === '--tag') out.tag = argv[++i];
    else if (a === '--channel') out.channel = argv[++i];
    else if (a === '--dry-run') out.dryRun = true;
  }
  return out;
}

function ghHeaders() {
  const h = { Accept: 'application/vnd.github+json', 'User-Agent': 'orrery-changelog' };
  if (process.env.GITHUB_TOKEN) h.Authorization = `Bearer ${process.env.GITHUB_TOKEN}`;
  return h;
}

async function gh(path) {
  const res = await fetch(`${API}${path}`, { headers: ghHeaders() });
  if (!res.ok) throw new Error(`GitHub API ${res.status} for ${path}: ${await res.text()}`);
  return res.json();
}

function semverKey(tag) {
  const m = /^v?(\d+)\.(\d+)\.(\d+)/.exec(tag);
  return m ? [Number(m[1]), Number(m[2]), Number(m[3])] : [0, 0, 0];
}

function cmpSemver(a, b) {
  const ka = semverKey(a);
  const kb = semverKey(b);
  for (let i = 0; i < 3; i++) if (ka[i] !== kb[i]) return ka[i] - kb[i];
  return 0;
}

async function listTags() {
  const tags = await gh(`/repos/${OWNER}/${REPO}/tags?per_page=100`);
  return tags.map((t) => t.name).filter((n) => /^v\d/.test(n)).sort(cmpSemver);
}

async function main() {
  const args = parseArgs(process.argv.slice(2));
  if (!args.summary) {
    console.error('error: --summary is required');
    process.exit(1);
  }

  const tags = await listTags(); // ascending semver
  const tag = args.tag || tags[tags.length - 1];
  if (!tag) throw new Error('no v* tags found');
  const idx = tags.indexOf(tag);
  const prevTag = idx > 0 ? tags[idx - 1] : null;

  let rawCommits;
  if (prevTag) {
    const cmp = await gh(`/repos/${OWNER}/${REPO}/compare/${prevTag}...${tag}`);
    rawCommits = cmp.commits; // oldest -> newest
  } else {
    const list = await gh(`/repos/${OWNER}/${REPO}/commits?sha=${tag}&per_page=30`);
    rawCommits = list.slice().reverse(); // make oldest -> newest
  }
  if (!rawCommits.length) throw new Error(`no commits found for ${tag}`);

  const head = rawCommits[rawCommits.length - 1];
  const commits = rawCommits
    .slice()
    .reverse() // newest -> oldest for display
    .map((c) => ({ sha: c.sha, subject: c.commit.message.split('\n')[0] }))
    .filter((c) => !isNoiseCommit(c.subject))
    .map((c) => {
      const p = parseConventionalCommit(c.subject);
      return { type: p.type, hash: c.sha, scope: p.scope, msg: p.msg };
    });

  const entry = buildReleaseEntry({
    tag,
    channel: args.channel,
    date: formatDate(head.commit.author.date),
    ref: head.sha,
    summary: args.summary,
    commits,
  });

  const file = resolve(dirname(fileURLToPath(import.meta.url)), '../../landing/changelog.html');
  const html = readFileSync(file, 'utf8');
  const next = injectRelease(html, entry);

  if (args.dryRun) {
    console.log(JSON.stringify([entry], null, 2));
    return;
  }
  writeFileSync(file, next);
  console.log(`changelog.html updated for ${tag} (${commits.length} commits).`);
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  main().catch((err) => {
    console.error(err.message || err);
    process.exit(1);
  });
}
```

## Task 5: Manual publish workflow

A `workflow_dispatch` workflow: run the generator with a required `summary`, commit the result back to `main`, then trigger the existing Render static site.

**Files:**
- Create: `.github/workflows/publish-changelog.yml`

- [ ] **Step 1: Create the workflow**

Create `.github/workflows/publish-changelog.yml`:

```yaml
name: Publish changelog

# Manually bake real commits for a release into the landing changelog, commit
# the result to main, and redeploy the Render static site. Manual-only because
# the summary is human-written copy. The commit-back uses the default
# GITHUB_TOKEN, whose pushes never retrigger a workflow — so no deploy loop.
on:
  workflow_dispatch:
    inputs:
      summary:
        description: "One-line summary for this release (marketing copy)"
        required: true
        type: string
      tag:
        description: "Release tag (vX.Y.Z). Blank = latest tag."
        required: false
        type: string
      channel:
        description: "Release channel badge"
        required: false
        default: "dev"
        type: string

permissions:
  contents: write

concurrency:
  group: publish-changelog
  cancel-in-progress: false

jobs:
  publish:
    runs-on: ubuntu-latest
    env:
      RENDER_API_KEY: ${{ secrets.RENDER_API_KEY }}
      SERVICE_NAME: orrery-landing
    steps:
      - uses: actions/checkout@v4
        with:
          ref: main

      - uses: actions/setup-node@v4
        with:
          node-version: "22"

      - name: Generate changelog entry
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: >
          node scripts/landing/gen-changelog.mjs
          --summary "${{ inputs.summary }}"
          ${{ inputs.tag && format('--tag {0}', inputs.tag) || '' }}
          --channel "${{ inputs.channel }}"

      - name: Commit regenerated changelog
        id: commit
        run: |
          set -euo pipefail
          if git diff --quiet -- landing/changelog.html; then
            echo "No changelog changes; nothing to commit."
            echo "changed=false" >> "$GITHUB_OUTPUT"
            exit 0
          fi
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add landing/changelog.html
          git commit -m "chore(landing): sync changelog for ${{ inputs.tag || 'latest' }} [skip ci]"
          git push origin HEAD:main
          echo "changed=true" >> "$GITHUB_OUTPUT"

      - name: Trigger Render deploy
        if: steps.commit.outputs.changed == 'true'
        run: |
          set -euo pipefail
          if [ -z "${RENDER_API_KEY:-}" ]; then
            echo "::error::RENDER_API_KEY is not set."
            exit 1
          fi
          API="https://api.render.com/v1"
          AUTH=(-H "Authorization: Bearer ${RENDER_API_KEY}" \
                -H "Accept: application/json" \
                -H "Content-Type: application/json")

          SID="$(curl -fsS "${AUTH[@]}" \
            "${API}/services?name=${SERVICE_NAME}&type=static_site&limit=1" \
            | jq -r '.[0].service.id // empty')"
          if [ -z "${SID}" ]; then
            echo "::error::static site '${SERVICE_NAME}' not found — run Deploy landing once first."
            exit 1
          fi

          DEPLOY="$(curl -fsS "${AUTH[@]}" -X POST \
            "${API}/services/${SID}/deploys" -d '{"clearCache":"do_not_clear"}')"
          DID="$(echo "${DEPLOY}" | jq -r '.id // empty')"
          echo "deploy id: ${DID:-<none>}"

          [ -z "${DID:-}" ] && { echo "Deploy started; no id to poll."; exit 0; }

          for i in $(seq 1 60); do
            STATUS="$(curl -fsS "${AUTH[@]}" \
              "${API}/services/${SID}/deploys/${DID}" | jq -r '.status // "unknown"')"
            echo "[$i] status: ${STATUS}"
            case "${STATUS}" in
              live) echo "::notice::Landing deployed — live."; exit 0 ;;
              build_failed|update_failed|pre_deploy_failed|canceled|deactivated)
                echo "::error::Deploy ended in '${STATUS}'."; exit 1 ;;
            esac
            sleep 10
          done
          echo "::warning::Timed out waiting for deploy ${DID}."
```

Note: this reuses the existing-service deploy path from `deploy-landing.yml:84-117` (kept self-contained so the working `deploy-landing.yml` is untouched). The `${{ inputs.tag && format(...) || '' }}` expression omits `--tag` entirely when the input is blank, so the generator defaults to the latest tag.
